# IMPORTANT: transformers is intentionally omitted. requirements.txt pins it to
# >=4.51,<4.52 (needed for Qwen3, model_type=qwen3). lmms-eval only declares

#
# Install order (see scripts/setup_eval_env.sh):
#   pip install -r requirements.txt
#   pip install -c <(echo "transformers>=4.51.0,<4.52.0") -r requirements-eval.txt
#   pip install -e third_party/lmms-eval --no-deps --no-build-isolation
